cmake_minimum_required(VERSION 3.24)
project(tradermade_backend LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Boost 1.81 CONFIG REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(CURL REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(Arrow CONFIG REQUIRED)
find_package(Parquet CONFIG REQUIRED)
find_package(Threads REQUIRED)

add_library(terminal_core src/market.cpp src/history.cpp src/config.cpp src/instruments.cpp)
target_include_directories(terminal_core PUBLIC include)
target_link_libraries(terminal_core PUBLIC nlohmann_json::nlohmann_json OpenSSL::Crypto
  CURL::libcurl Arrow::arrow_shared Parquet::parquet_shared Threads::Threads)
target_compile_definitions(terminal_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
file(TO_CMAKE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/.." TERMINAL_ROOT)
target_compile_definitions(terminal_core PUBLIC TM_PROJECT_ROOT="${TERMINAL_ROOT}")

add_executable(tradermade_backend main.cpp src/network.cpp)
target_link_libraries(tradermade_backend PRIVATE terminal_core Boost::headers OpenSSL::SSL)
if(WIN32)
  target_link_libraries(tradermade_backend PRIVATE ws2_32 mswsock crypt32)
endif()
if(MSVC)
  target_compile_options(terminal_core PRIVATE /W4 /utf-8)
  target_compile_options(tradermade_backend PRIVATE /W4 /utf-8 /bigobj)
else()
  target_compile_options(terminal_core PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(tradermade_backend PRIVATE -Wall -Wextra -Wpedantic)
  if(MINGW)
    target_compile_options(tradermade_backend PRIVATE -Wa,-mbig-obj)
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(terminal_tests tests/core_tests.cpp)
  target_link_libraries(terminal_tests PRIVATE terminal_core)
  add_test(NAME terminal_core COMMAND terminal_tests)
endif()
